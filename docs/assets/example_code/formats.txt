Use "../../../src/VoltScriptLogging"

Sub Initialize

    Dim stringFormat as New BaseLogWriter("stringFormat", LOG_DEBUG, LOG_FATAL, "{{LEVELNAME}}: {{MESSAGE}}")
    Dim jsonFormat  as New BaseLogWriter("jsonFormat", LOG_DEBUG, LOG_FATAL, |{
        "level": "{{LEVELNAME}}",
        "message": "{{MESSAGE}}",
        "extInfo": "{{EXTINFO}}",
        "stack": "{{STACKTRACE}}"
    }|)
    
    Call globalLogSession.createLogEntry(LOG_DEBUG, "Entered Initialize", "", Nothing)

    Call globalLogSession.addLogWriter(stringFormat)
    Call globalLogSession.addLogWriter(jsonFormat)

    Call globalLogSession.createLogEntry(LOG_INFO, "Added log writers", "string and json", Nothing)
    Call globalLogSession.createLogEntry(LOG_DEBUG, "Finished Initialize", "", Nothing)

End Sub

Sub Terminate

    Call globalLogSession.createLogEntry(LOG_DEBUG, "Entered Terminate", "", Nothing)
    Call globalLogSession.createLogEntry(LOG_INFO, "Terminating", "Note this fires before Delete in VoltScriptLogging", Nothing)
    Call globalLogSession.createLogEntry(LOG_DEBUG, "Finished Terminate", "", Nothing)

End Sub
